Simple Ansible Example for Infrastructure

Create a file named deploy_infrastructure.yml:
---
- name: Deploy infrastructure for microservices
  hosts: your_server  # Replace with your server IP or hostname
  become: yes
  tasks:
    - name: Install Docker
      apt:
        name: docker.io
        state: present
        update_cache: yes

    - name: Pull and run Product Service
      docker_container:
        name: product-service
        image: yourusername/product-service:latest
        ports: "5000:5000"
        restart_policy: always

    - name: Pull and run Dealer Service
      docker_container:
        name: dealer-service
        image: yourusername/dealer-service:latest
        ports: "8080:8080"
        restart_policy: always

    - name: Pull and run Frontend
      docker_container:
        name: frontend
        image: yourusername/frontend:latest
        ports: "80:80"  # Assuming frontend runs on port 80
        restart_policy: always
